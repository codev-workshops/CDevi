+import { redis } from '../lib/redis';
+
+const WINDOW_MS = 60_000;
+const LIMIT = 100;
+
+export async function rateLimit(key: string, now = Date.now()) {
+ const bucket = `rl:${key}`;
+ const tx = redis.multi();
+ tx.zremrangebyscore(bucket, 0, now - WINDOW_MS);
+ tx.zadd(bucket, now, `${now}`);
+ tx.zcard(bucket);
+ tx.pexpire(bucket, WINDOW_MS);
+ const [, , count] = await tx.exec();
+ const allowed = Number(count) <= LIMIT;
+ return { allowed, retryAfterMs: allowed ? 0 : WINDOW_MS };
+}
