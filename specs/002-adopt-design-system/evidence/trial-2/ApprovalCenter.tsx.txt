/* ---------- Sample data (specs/001 US2, acceptance scenario 1) ---------- */

type PendingKind = 'approval' | 'clarification';

interface PendingItem {
  /** Requirement / workflow identifier. */
  id: string;
  kind: PendingKind;
  /** Short label of the request type, e.g. "Approve PR merge". */
  request: string;
  /** The exact thing being asked of the person. */
  ask: string;
  /** Who or which agent raised it. */
  requester: string;
  /** Agent run that raised it, when applicable. */
  run?: string;
  /** Workflow stage the item belongs to. */
  stage: string;
  requestedAt: string;
  risk: RiskLevel;
  href: string;
}

/* ---------- Ordering: highest risk first, then oldest first ---------- */

const RISK_RANK: Record<RiskLevel, number> = { CRITICAL: 0, HIGH: 1, MEDIUM: 2, LOW: 3 };

export function sortPending(items: readonly PendingItem[]): PendingItem[] {
  return [...items].sort((a, b) => {
    const byRisk = RISK_RANK[a.risk] - RISK_RANK[b.risk];
    if (byRisk !== 0) return byRisk;
    return Date.parse(a.requestedAt) - Date.parse(b.requestedAt);
  });
}

/* ---------- Formatting ---------- */

function relativeTime(iso: string, now: Date = NOW): string {
  const minutes = Math.max(0, Math.round((now.getTime() - Date.parse(iso)) / 60_000));
  if (minutes < 1) return 'just now';
  if (minutes < 60) return `${minutes} min ago`;
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  return m === 0 ? `${h} h ago` : `${h} h ${m} min ago`;
}

function absoluteTime(iso: string): string {
  return new Intl.DateTimeFormat('en-GB', {
    day: '2-digit',
    month: 'short',
    hour: '2-digit',
    minute: '2-digit',
    timeZone: 'UTC',
    timeZoneName: 'short',
  }).format(new Date(iso));
}

function kindLabel(kind: PendingKind): string {
  return kind === 'approval' ? 'approval' : 'clarification';
}
